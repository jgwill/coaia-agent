Title: accepted_redirection_hook.sh
Description: #!/bin/bash
# Accepted Redirection Hook
# Captures when user accepts a tool (Yes+Tab) and provides follow-up instructions
# Mirror of corrective_feedback_hook.sh — this is the "Yes with next task" pattern
# Runs on UserPromptSubmit and parses transcript for accepted tool_results with text

input=$(cat -)
session_id=$(echo "$input" | jq -r .session_id)

# Determine base directory
if [ -d "/workspace/coaia-agent/.hch/sessions" ]; then
    base_dir="/src"
else
    script_dir="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
    base_dir="$(cd "$script_dir/../.." && pwd)"
fi

output_dir="/workspace/coaia-agent/.hch/sessions/$session_id"
mkdir -p "$output_dir"

# Parse transcript for accepted tool_results paired with follow-up text
transcript_file=$(echo "$input" | jq -r '.transcript_path // empty')

if [ -f "$transcript_file" ]; then
    # Check recent transcript entries (last 30)
    tail -30 "$transcript_file" 2>/dev/null | while IFS= read -r line; do
        msg_type=$(echo "$line" | jq -r '.type' 2>/dev/null)
        [ "$msg_type" != "user" ] && continue

        # Check if this user message has BOTH:
        # 1) A tool_result with is_error NOT true (accepted)
        # 2) A text content block (the follow-up instruction)
        has_accepted_tool_result=$(echo "$line" | jq -r '
            [.message.content[] | select(.type == "tool_result" and (.is_error == null or .is_error == false))] | length > 0
        ' 2>/dev/null)

        has_text=$(echo "$line" | jq -r '
            [.message.content[] | select(.type == "text" and (.text | length > 5))] | length > 0
        ' 2>/dev/null)

        if [ "$has_accepted_tool_result" = "true" ] && [ "$has_text" = "true" ]; then
            # Extract the tool_use_id from the accepted tool_result
            tool_use_id=$(echo "$line" | jq -r '
                .message.content[] | select(.type == "tool_result" and (.is_error == null or .is_error == false)) | .tool_use_id
            ' 2>/dev/null | head -1)

            # Extract the follow-up instruction text
            followup_text=$(echo "$line" | jq -r '
                [.message.content[] | select(.type == "text") | .text] | join("\n")
            ' 2>/dev/null)

            msg_timestamp=$(echo "$line" | jq -r '.timestamp // empty' 2>/dev/null)

            if [ -n "$tool_use_id" ] && [ -n "$followup_text" ]; then
                # Deduplicate: check if we already captured this tool_use_id
                already_captured=false
                if [ -f "$output_dir/_claude_AcceptedRedirection.jsonl" ]; then
                    if grep -q "\"accepted_tool_use_id\":\"$tool_use_id\"" "$output_dir/_claude_AcceptedRedirection.jsonl" 2>/dev/null; then
                        already_captured=true
                    fi
                fi

                if [ "$already_captured" = false ]; then
                    # Look up what tool was accepted from PreToolUse log
                    pretool_file="$output_dir/_claude_PreToolUse.jsonl"
                    tool_name="unknown"
                    tool_input="{}"

                    if [ -f "$pretool_file" ]; then
                        tool_proposal=$(grep "\"tool_use_id\":\"$tool_use_id\"" "$pretool_file" | tail -1)
                        if [ -n "$tool_proposal" ]; then
                            tool_name=$(echo "$tool_proposal" | jq -r '.tool_name // "unknown"')
                            tool_input=$(echo "$tool_proposal" | jq -c '.tool_input // {}')
                        fi
                    fi

                    timestamp=$(date -Iseconds)

                    # Build accepted redirection JSON
                    redirection=$(jq -n \
                        --arg session_id "$session_id" \
                        --arg hook_event_name "AcceptedRedirection" \
                        --arg accepted_tool_use_id "$tool_use_id" \
                        --arg accepted_tool_name "$tool_name" \
                        --argjson accepted_tool_input "$tool_input" \
                        --arg followup_instruction "$followup_text" \
                        --arg acceptance_timestamp "$msg_timestamp" \
                        --arg timestamp "$timestamp" \
                        '{
                            session_id: $session_id,
                            hook_event_name: $hook_event_name,
                            accepted_tool_use_id: $accepted_tool_use_id,
                            accepted_tool_name: $accepted_tool_name,
                            accepted_tool_input: $accepted_tool_input,
                            followup_instruction: $followup_instruction,
                            acceptance_timestamp: $acceptance_timestamp,
                            timestamp: $timestamp
                        }')

                    # Append to redirection log
                    echo "$redirection" >> "$output_dir/_claude_AcceptedRedirection.jsonl"
                    echo "$redirection" > "$output_dir/last_claude_AcceptedRedirection.json"

                    echo "Captured accepted redirection: $tool_name ($tool_use_id) with follow-up at $(date '+%Y-%m-%d %H:%M:%S')" >> "$output_dir/trace.log"
                fi
            fi
        fi
    done
fi

exit 0
